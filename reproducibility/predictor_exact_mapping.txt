def build_rich_eq_context(eq, idx=None):
    """
    v23核心改动：智能过滤全字段传入LLM。
    策略：去掉null值 + 去掉纯技术字段（parameters/equation_chain/id等）
    其余所有有值字段全部保留，让LLM自己判断哪些有用。
    
    字段映射说明（数据库真实字段）：
      location        = applicable_region（适用地区）
      source          = reference（参考文献）
      r_squared       = R²（仅22.5%填充，空时用quality_score代替）
      quality_score   = 综合质量分（100%填充，0.7-0.85）
      size_class      = 适用范围（含径级/林龄/说明文字，97.7%填充）
    """
    # 纯技术字段：对LLM适用性判断无意义，过滤掉
    SKIP_FIELDS = {'id', 'source_dataset', 'parameters', 'equation_chain',
                   'variables'}  # variables可从equation推断

    parts = []
    if idx is not None:
        parts.append(f'[候选方程{idx}]')

    # 字段中文别名映射（让LLM更容易理解字段含义）
    FIELD_ALIAS = {
        'tree_type':        '树种',
        'component':        '组分',
        'equation':         '公式',
        'location':         '适用地区',      # = applicable_region
        'size_class':       '适用范围',       # 含径级/林龄
        'quality_score':    '综合质量分',
        'r_squared':        'R²',
        'sample_size':      '样本量n',
        'source':           '参考文献',       # = reference
        'authors':          '作者',
        'data_quality':     '数据质量',
        'tags':             '标签',
        'component_original': '原始组分描述',
        'stand_age':        '林龄',
        'forest_type':      '林分类型',
        'stand_origin':     '林分起源',
        'dominant_species': '优势种学名',
        'altitude':         '海拔',
        'equation_id':      '方程编号',
        'equation_name':    '方程名称',
        'publication_year': '发表年份',
    }

    for key, val in eq.items():
        if key in SKIP_FIELDS:
            continue
        # 过滤空值：None / 空字符串 / 空列表 / 0值样本量
        if val is None or val == '' or val == []:
            continue
        if key == 'sample_size' and (val == 0 or str(val) == '0'):
            continue
        # r_squared 有时是空字符串
        if key == 'r_squared' and str(val).strip() == '':
            continue

        label = FIELD_ALIAS.get(key, key)

        # 特殊格式化
        if key == 'tags' and isinstance(val, list):
            parts.append(f'{label}: {", ".join(str(t) for t in val)}')
        elif key == 'quality_score':
            try:
                parts.append(f'{label}: {round(float(val), 3)}（0-1分，越高越好）')
            except:
                parts.append(f'{label}: {val}')
        elif key == 'parameters':
            pass  # 已在SKIP_FIELDS中过滤
        else:
            val_str = str(val)
            # 过长文本截断
            if len(val_str) > 150:
                val_str = val_str[:147] + '...'
            parts.append(f'{label}: {val_str}')

    # [消融开关] Ablation-2: 切除完整元数据，只传基础4字段
    if not ENABLE_FULL_META:
        basic = []
        if idx is not None: basic.append(f'[方程{idx}]')
        basic.append(f'公式: {eq.get("equation","")}')
        basic.append(f'树种: {eq.get("tree_type","")}')
        basic.append(f'地区: {eq.get("location","")}')
        basic.append(f'组分: {eq.get("component","")}')
        return ' | '.join(basic)
    return ' | '.join(parts)

def check_diameter_type(slots: dict, top_candidates: list) -> dict | None:
    """
    若用户提供的是基径(D0)，但候选 Top1 方程只使用胸径(DBH/D)，
    返回拒绝信号字典；否则返回 None（不拒绝）。

    判断逻辑：
      1. diameter_type == "基径"
      2. 方程的 variables/equation/tags 里不含 D0/d0/基径/地径
      3. → 返回提示信息
    """
    if not slots or not top_candidates:
        return None
    dtype = slots.get('diameter_type', '未知')
    if dtype != '基径':
        return None

    top_eq   = top_candidates[0]
    eq_str   = str(top_eq.get('equation',  '') or '').lower()
    var_str  = str(top_eq.get('variables', '') or '').lower()
    tags_str = str(top_eq.get('tags',      '') or '').lower()
    comp_str = str(top_eq.get('component_original', '') or '').lower()

    # 方程本身就用 D0/d0/基径 → 匹配，不拒绝
    basal_indicators = ['d0', 'd_0', '基径', '地径', '根径']
    if any(ind in eq_str or ind in var_str or ind in tags_str or ind in comp_str
           for ind in basal_indicators):
        return None

    # 方程只用 D（胸径）但用户给的是基径 → 提示
    eq_display = top_eq.get('equation', '')[:50]
    return {
        'intent': 'diameter_mismatch',
        'reason': 'diameter_mismatch',
        'message': (
            f"您提供的是基径（地面处直径），"
            f"但该方程（{eq_display}）需要的是胸径（DBH，1.3m处直径）。"
            f"两者通常相差15~30%，直接代入会产生较大误差。"
            f"建议重新测量胸径后再计算，或查找使用基径的方程。"
        )
    }

def calculate_with_chain(
    formula: str,
    extracted: ExtractedParams,
    matcher: EnhancedEquationMatcher,
    match_info: Dict,
    verbose: bool = False
) -> Optional[float]:
    is_chain, required_components = detect_chain_equation(formula)
    
    if not is_chain:
        return calc_equation_single(formula, extracted)
    
    if verbose:
        print(f"  🔗 检测到推理链式方程")
        print(f"     需要先计算组分: {', '.join(required_components)}")
    
    component_values = {}
    
    for component in required_components:
        if verbose:
            print(f"\n  [子任务] 计算 {component} 组分生物量...")
        
        component_params = ExtractedParams(
            species=extracted.species,
            genus=extracted.genus,
            region=extracted.region,
            province=extracted.province,
            macro_region=extracted.macro_region,
            component=component,
            D=extracted.D,
            H=extracted.H,
            D0=extracted.D0,
            V=extracted.V,
            CW=extracted.CW,
            age=extracted.age
        )
        
        candidates = []
        for eq in matcher.equations:
            eq_component = normalize_component(get_field(eq.raw_data, 'Component', ''))
            if eq_component == component:
                candidates.append(eq)
        
        if not candidates:
            if verbose:
                print(f"    ⚠️  未找到 {component} 组分的方程，跳过")
            continue
        
        scored_eqs = []
        for eq in candidates[:100]:
            score = matcher.calculate_score_with_match_info(eq.raw_data, match_info, component_params)
            scored_eqs.append((score, eq))
        
        scored_eqs.sort(key=lambda x: x[0], reverse=True)
        
        if scored_eqs:
            best_score, best_eq = scored_eqs[0]
            best_formula = get_field(best_eq.raw_data, 'Formula', '')
            
            if verbose:
                print(f"    ✅ 选择方程: {best_formula[:60]}... (评分={best_score:.1f})")
            
            component_result = calc_equation_single(best_formula, component_params)
            
            if component_result is not None:
                component_values[f'W_{component}'] = component_result
                if verbose:
                    print(f"    ✅ {component} 组分 = {component_result:.4f} kg")
            else:
                if verbose:
                    print(f"    ❌ {component} 组分计算失败")
    
    if verbose:
        print(f"\n  [最终计算] 使用组分值计算总量...")
        print(f"     组分值: {component_values}")
    
    final_result = calc_equation_single(
        formula,
        extracted,
        W_gan=component_values.get('W_干'),
        W_zhi=component_values.get('W_枝'),
        W_ye=component_values.get('W_叶')
    )
    
    return final_result
