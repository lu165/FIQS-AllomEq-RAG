def find_similar_trees_in_province(
    target_tree: str, 
    location: str,
    equations: List[Dict],
    bge_model: SentenceTransformer,
    component: str = None
) -> List[Dict]:
    """
    在同省/同地区内，找生态相似的树种方程
    
    步骤：
    1. 筛选同省方程
    2. BGE计算树种名称相似度
    3. LLM验证生态相似性
    4. 返回可用方程列表
    """
    
    # 步骤1：筛选同省/同地区方程
    province_equations = []
    for eq in equations:
        eq_loc = eq.get('location', '')
        # 宽松匹配：用户地名在方程地名中，或方程地名在用户地名中
        if location in eq_loc or eq_loc in location:
            province_equations.append(eq)
    
    if not province_equations:
        print(f"[树种泛化] {location} 无任何方程，无法泛化")
        return []
    
    # 步骤2：BGE计算树种相似度
    target_vec = bge_model.encode([target_tree], normalize_embeddings=True)[0]
    
    candidate_trees = {}  # tree_name -> (相似度, 方程列表)
    for eq in province_equations:
        eq_tree = eq.get('tree_type', '')
        if eq_tree == target_tree:
            continue  # 跳过自身
        
        # 组分过滤（可选）
        if component and eq.get('component') != component:
            if component not in eq.get('component', ''):
                continue
        
        if eq_tree not in candidate_trees:
            tree_vec = bge_model.encode([eq_tree], normalize_embeddings=True)[0]
            sim = float(np.dot(target_vec, tree_vec))
            candidate_trees[eq_tree] = {'similarity': sim, 'equations': []}
        
        candidate_trees[eq_tree]['equations'].append(eq)
    
    # 按相似度排序，取前5个候选树种
    sorted_trees = sorted(
        candidate_trees.items(),
        key=lambda x: x[1]['similarity'],
        reverse=True
    )[:5]
    
    print(f"\n[树种泛化] BGE候选树种（同省）：")
    for tree, info in sorted_trees:
        print(f"  {tree}: 相似度={info['similarity']:.3f}")
    
    # 步骤3：LLM验证生态相似性
    valid_results = []
    for tree_name, info in sorted_trees:
        can_gen, reason = check_ecological_similarity(
            target_tree, tree_name, location
        )
        if can_gen:
            print(f"  ✅ {tree_name}: {reason}")
            for eq in info['equations']:
                eq['_generalize_reason'] = f"原始树种'{target_tree}'在{location}无方程，使用生态相似树种'{tree_name}'替代。{reason}"
                eq['_original_tree'] = target_tree
                eq['_substitute_tree'] = tree_name
                valid_results.append(eq)
        else:
            print(f"  ❌ {tree_name}: {reason}")
    
    return valid_results

def get_location_geographic_info(location: str) -> Dict:
    """
    用LLM获取地名的地理信息（坐标、省份、生态区等）
    """
    prompt = f"""请提供"{location}"的地理信息，返回JSON格式。

{{
  "name": "标准名称",
  "province": "所属省份",
  "city": "所属城市（如有）",
  "lat": 纬度(小数),
  "lon": 经度(小数),
  "climate_zone": "气候带（如：温带季风、亚热带季风等）",
  "vegetation_zone": "植被带（如：温带针叶林、亚热带常绿阔叶林等）",
  "altitude_range": "海拔范围（如：200-500m）",
  "description": "一句话描述该地区生态特征"
}}

注意：只返回JSON，不要其他文字。"""

    result = call_llm(prompt, temperature=0.1)
    
    try:
        import re
        json_match = re.search(r'\{.*\}', result, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
    except:
        pass
    
    return {"name": location, "province": "", "lat": None, "lon": None}

def calculate_geographic_distance(loc1_info: Dict, loc2_info: Dict) -> float:
    """
    计算两个地点的地理距离（km），使用Haversine公式
    """
    import math
    
    lat1 = loc1_info.get('lat')
    lon1 = loc1_info.get('lon')
    lat2 = loc2_info.get('lat')
    lon2 = loc2_info.get('lon')
    
    if any(v is None for v in [lat1, lon1, lat2, lon2]):
        return float('inf')
    
    R = 6371  # 地球半径km
    lat1, lon1, lat2, lon2 = map(math.radians, [lat1, lon1, lat2, lon2])
    
    dlat = lat2 - lat1
    dlon = lon2 - lon1
    a = math.sin(dlat/2)**2 + math.cos(lat1)*math.cos(lat2)*math.sin(dlon/2)**2
    c = 2 * math.asin(math.sqrt(a))
    
    return R * c

def llm_rank_locations_by_distance(
    query_location: str,
    query_geo: Dict,
    candidate_locations: List[str],
    tree_type: str = None,
    component: str = None
) -> List[Dict]:
    """
    用LLM综合判断候选地区与查询地区的相似度
    同时考虑：地理距离、气候相似性、植被带相似性
    
    返回：按综合相似度排序的地区列表
    """
    
    # 🔥 按精确度分层：子地区 > 省级 > 跨省泛指
    def location_specificity(loc):
        """精确度评分：越具体分越高"""
        if any(k in loc for k in ['林场','保护区','实验区','林业局','林区']):
            return 4  # 林场级
        if any(k in loc for k in ['县','市','山','岭','坡','流域']):
            return 3  # 县市/山区级
        if any(k in loc for k in ['省','自治区','全省']):
            return 2  # 省级
        if any(k in loc for k in ['东北','华南','西南','东三省','全国']):
            return 1  # 跨省泛指
        return 2  # 默认省级
    
    # 分层展示给LLM（精确地区排前面）
    sorted_candidates = sorted(
        candidate_locations[:20],
        key=location_specificity,
        reverse=True
    )
    candidates_text = "\n".join([
        f"{i+1}. {loc}（精确度:{'高' if location_specificity(loc)>=3 else '中' if location_specificity(loc)==2 else '低'}）"
        for i, loc in enumerate(sorted_candidates)
    ])
    
    prompt = f"""你是一个地理和林业专家。

查询地区：{query_location}
所属省份：{query_geo.get('province','未知')}
气候带：{query_geo.get('climate_zone','未知')}
植被带：{query_geo.get('vegetation_zone','未知')}

候选地区（已按精确度排序）：
{candidates_text}

任务：选出与"{query_location}"地理空间位置最近的1个地区。

选择原则：
1. 优先选精确度高的（如具体山区、林场 > 省级 > 跨省泛指）
2. 地理距离最近（同省同山系 > 同省不同山系 > 邻省）
3. 气候植被类型相似
{f'4. 有{tree_type}分布记录' if tree_type else ''}

重要：如果有精确度高且距离近的子地区，优先选它，不要选省级泛指。

返回JSON（只选1个）：
{{
  "rankings": [
    {{
      "rank": 1,
      "location": "地区名（必须与候选列表完全一致，不含精确度标注）",
      "distance_km": 估算直线距离(km),
      "similarity_score": 0-1,
      "reason": "理由：距离+气候+树种"
    }}
  ]
}}"""

    result = call_llm(prompt, temperature=0.1)
    
    try:
        import re
        json_match = re.search(r'\{.*\}', result, re.DOTALL)
        if json_match:
            data = json.loads(json_match.group())
            return data.get('rankings', [])
    except Exception as e:
        print(f"[地区泛化] LLM解析失败: {e}")
    
    return []

def find_nearest_location_in_db(
    query_location: str,
    equations: List[Dict],
    tree_type: str = None,
    component: str = None,
    max_distance_km: float = 500
) -> List[Dict]:
    """
    地区泛化主函数
    
    步骤：
    1. LLM获取查询地区的地理信息
    2. 从方程库提取候选地区（优先同省）
    3. LLM综合判断地理+生态相似度
    4. 返回按相似度排序的方程列表
    
    Args:
        query_location: 用户查询的地区
        equations: 方程库
        tree_type: 树种（用于过滤候选地区）
        component: 组分（用于过滤候选地区）
        max_distance_km: 最大允许距离
    
    Returns:
        匹配的方程列表（含泛化说明）
    """
    
    print(f"\n[地区泛化] 查询地区: {query_location}")
    print(f"[地区泛化] 树种: {tree_type}, 组分: {component}")
    
    # Step 1: 获取查询地区地理信息
    print("[地区泛化] Step1: 获取地理信息...")
    query_geo = get_location_geographic_info(query_location)
    print(f"  省份: {query_geo.get('province')}")
    print(f"  坐标: ({query_geo.get('lat')}, {query_geo.get('lon')})")
    print(f"  气候带: {query_geo.get('climate_zone')}")
    print(f"  植被带: {query_geo.get('vegetation_zone')}")
    
    # Step 2: 从方程库提取候选地区
    print("[地区泛化] Step2: 提取候选地区...")
    
    # 提取所有地区
    all_locations = list(set(
        eq.get('location', '').strip()
        for eq in equations
        if eq.get('location', '').strip()
    ))
    
    # 优先策略：同省地区
    province = query_geo.get('province', '')
    same_province_locs = []
    other_locs = []
    
    for loc in all_locations:
        if province and province[:2] in loc:  # 省份简称匹配
            same_province_locs.append(loc)
        else:
            other_locs.append(loc)
    
    # 如果有树种过滤，进一步筛选
    if tree_type:
        def has_tree_in_location(loc, tree):
            return any(
                tree in eq.get('tree_type', '') or 
                eq.get('tree_type', '') in tree
                for eq in equations
                if eq.get('location', '') == loc
            )
        
        same_province_with_tree = [
            loc for loc in same_province_locs 
            if has_tree_in_location(loc, tree_type)
        ]
        
        if same_province_with_tree:
            candidate_locations = same_province_with_tree + [
                l for l in same_province_locs 
                if l not in same_province_with_tree
            ]
        else:
            # 同省没有该树种，扩展到全库
            print(f"  同省无'{tree_type}'方程，扩展到全库搜索")
            candidate_locations = all_locations
    else:
        candidate_locations = same_province_locs + other_locs
    
    print(f"  候选地区数: {len(candidate_locations)}")
    print(f"  同省地区: {same_province_locs[:5]}")
    
    # Step 3: LLM综合判断相似度
    print("[地区泛化] Step3: LLM地理相似度判断...")
    rankings = llm_rank_locations_by_distance(
        query_location=query_location,
        query_geo=query_geo,
        candidate_locations=candidate_locations,
        tree_type=tree_type,
        component=component
    )
    
    if not rankings:
        print("  ⚠️ LLM排序失败，使用同省第一个地区")
        if same_province_locs:
            rankings = [{"location": same_province_locs[0], 
                        "similarity_score": 0.5,
                        "reason": "同省降级匹配",
                        "distance_km": 999}]
    
    print(f"\n  LLM地区排序结果:")
    for r in rankings[:3]:
        print(f"    {r.get('rank','?')}. {r.get('location')} "
              f"(距离~{r.get('distance_km','?')}km, "
              f"相似度:{r.get('similarity_score','?'):.2f})")
        print(f"       理由: {r.get('reason','')}")
    
    # Step 4: 收集匹配方程
    print("\n[地区泛化] Step4: 收集匹配方程...")
    result_equations = []
    
    for rank_info in rankings:
        matched_loc = rank_info.get('location', '')
        similarity = rank_info.get('similarity_score', 0)
        distance_km = rank_info.get('distance_km', 999)
        reason = rank_info.get('reason', '')
        
        # 距离过滤
        if distance_km != 999 and distance_km > max_distance_km:
            print(f"  ⚠️ {matched_loc} 距离{distance_km}km超过{max_distance_km}km，跳过")
            continue
        
        # 收集该地区的方程（🔥 用省份等价匹配，黑龙江=黑龙江省）

            if are_same_province(eq.get('location','').strip(), matched_loc)
        ]

            # 精确匹配

                eq for eq in loc_equations

            }

            ]

                loc_equations = comp_filtered

            eq_copy = eq.copy()
            eq_copy['_location_generalized'] = True

            eq_copy['_distance_km'] = distance_km

        if len(result_equations) >= 5:
            break
    
    print(f"\n[地区泛化] ✅ 共找到 {len(result_equations)} 个泛化方程")

def _call_shared_model(self, prompt: str) -> str:
        """✅ 调用共享模型"""
        if not self.use_local_model or self.model is None or self.tokenizer is None:
            print("    [TreeMatcher] ⚠️ 模型未加载，跳过")
            raise Exception("共享模型未加载")
        
        print(f"  [TreeMatcher] 发送请求...")
        
        try:
            # 构建消息
            messages = [
                {
                    "role": "system",
                    "content": "你是一位专业的林业分类学专家，精通树种分类和生态学。你的回答必须是有效的JSON格式，不要有markdown标记。"

                    "role": "user",
                    "content": prompt
                }
            ]
            
            # 使用 tokenizer 的 apply_chat_template
            text = self.tokenizer.apply_chat_template(
                messages,
                tokenize=False,
                add_generation_prompt=True
            )
            
            # ✅ 使用共享的 device
            model_inputs = self.tokenizer(
                [text], 
                return_tensors="pt",
                padding=True,
                truncation=True,
                max_length=2048
            ).to(self.device)
            
            # 生成
            with torch.no_grad():
                generated_ids = self.model.generate(
                    **model_inputs,
                    max_new_tokens=800,
                    temperature=0.3,
                    do_sample=True,
                    top_p=0.9,
                    repetition_penalty=1.1,
                    pad_token_id=self.tokenizer.pad_token_id,
                    eos_token_id=self.tokenizer.eos_token_id,
                )
            
            # 解码输出
            generated_ids = [
                output_ids[len(input_ids):] 
                for input_ids, output_ids in zip(model_inputs.input_ids, generated_ids)
            ]
            response = self.tokenizer.batch_decode(generated_ids, skip_special_tokens=True)[0]
            
            print(f"  [TreeMatcher] ✅ 成功，响应长度: {len(response)}")
            
            return response.strip()
        
        except Exception as e:
            raise Exception(f"共享模型调用失败: {str(e)}")

def _fallback_rule_based_matching(self, target_species: str, available_species: List[str]) -> Dict:
        """
        回退的规则匹配方法（当模型调用失败时）
        
        Args:
            target_species: 目标树种
            available_species: 可用树种列表
        
        Returns:
            匹配结果
        """
        print(f"  [规则匹配] 使用基于规则的简单匹配...")
        
        # 规则1: 字符串相似度匹配
        candidates = []
        
        for species in available_species:
            # 计算简单的字符重叠度
            common_chars = set(target_species) & set(species)
            similarity = len(common_chars) / max(len(set(target_species)), len(set(species)))
            
            # 同科属加分（简单规则）
            if ('松' in target_species and '松' in species) or \
               ('杉' in target_species and '杉' in species) or \
               ('桉' in target_species and '桉' in species):
                similarity += 0.3
            
            candidates.append({
                'species': species,
                'similarity': similarity
            })
        
        # 排序
        candidates.sort(key=lambda x: x['similarity'], reverse=True)
        
        if candidates:
            best = candidates[0]
            return {
                'species': best['species'],
                'direct_match': False,
                'similarity_score': min(best['similarity'], 0.8),
                'reason': '基于规则的简单匹配',
                'confidence': 'low'
            }
        else:
            return {
                'species': None,
                'direct_match': False,
                'reason': '规则匹配失败'
            }

        """使用共享模型进行地理推理"""
        if not self.use_local_model or self.model is None or self.tokenizer is None:
            if self.verbose:
                print("    [地区匹配器] ⚠️ 模型未加载，跳过")
            return ""
        
        for attempt in range(max_retries):
            try:
                if self.verbose:
                    print(f"    [地区匹配器] 第{attempt+1}次尝试...")
                
                messages = [
                    {
                        "role": "system", 
                        "content": "你是一个中国地理专家，精通各省市县、山脉、林场、自然保护区的地理位置关系。请严格按照要求输出JSON格式。"
                    },
                    {
                        "role": "user", 
                        "content": prompt
                    }
                ]
                
                text = self.tokenizer.apply_chat_template(
                    messages,
                    tokenize=False,
                    add_generation_prompt=True
                )
                
                model_inputs = self.tokenizer(
                    [text], 
                    return_tensors="pt",
                    padding=True,
                    truncation=True,
                    max_length=2048
                ).to(self.device)
                
                with torch.no_grad():
                    generated_ids = self.model.generate(
                        **model_inputs,
                        max_new_tokens=512,
                        temperature=0.3,
                        do_sample=True,
                        top_p=0.9,
                        repetition_penalty=1.1,
                        pad_token_id=self.tokenizer.pad_token_id,
                        eos_token_id=self.tokenizer.eos_token_id,
                    )
                
                generated_ids = [
                    output_ids[len(input_ids):] 
                    for input_ids, output_ids in zip(model_inputs.input_ids, generated_ids)
                ]
                response = self.tokenizer.batch_decode(generated_ids, skip_special_tokens=True)[0]
                
                if self.verbose:
                    print(f"    [地区匹配器] ✅ 成功 (长度: {len(response)})")
                return response.strip()
                
            except Exception as e:
                print(f"    [地区匹配器] ❌ 异常: {e}")
                if attempt < max_retries - 1:
                    import time
                    time.sleep(1)
                    continue
                else:
                    raise
        
        return ""

                if not self._would_form_cycle(child_location, parent, hierarchy):

                    skipped_count += 1

            'level': 'country',
            'children': [loc for loc in all_locations if hierarchy[loc]['parent'] is None],
            'parent': None,
            'has_equations': has_national
        }
        
        for loc in hierarchy['全国']['children']:
            if hierarchy[loc]['parent'] is None:
                hierarchy[loc]['parent'] = '全国'
        
        hierarchy['全国']['parent'] = None
        
        self._validate_hierarchy(hierarchy)
        self._print_hierarchy_summary(hierarchy)
        
        return hierarchy

def _would_form_cycle(self, child: str, parent: str, hierarchy: Dict) -> bool:
        """检查添加父子关系是否会形成循环"""
        if child == parent:
            return True
        
        visited = set()
        current = parent
        max_depth = 50
        
        while current and current in hierarchy:
            if current == child:
                return True
            
            if current in visited:
                return True
            
            visited.add(current)
            current = hierarchy[current].get('parent')
            
            if len(visited) > max_depth:
                return True
        
        return False

def _find_parent_location(self, child_location: str, all_locations: Set[str]) -> Optional[str]:
        """为子地区找到父地区"""
        child_level = self._determine_level(child_location)
        
        level_priority = {
            'country': 0,
            'province': 1,
            'city': 2,
            'county': 3,
            'forest': 4,
            'other': 5
        }
        
        child_priority = level_priority.get(child_level, 5)
        candidates = []
        
        for potential_parent in all_locations:
            if potential_parent == child_location:
                continue
            
            if len(potential_parent) >= len(child_location):
                if not (potential_parent in child_location and potential_parent != child_location):
                    continue
            
            parent_level = self._determine_level(potential_parent)
            parent_priority = level_priority.get(parent_level, 5)
            
            if parent_priority >= child_priority:
                continue
            
            if potential_parent in child_location and potential_parent != child_location:
                candidates.append({
                    'location': potential_parent,
                    'priority': parent_priority,
                    'length': len(potential_parent),
                    'score': 100
                })
                continue
            
            if child_level in ['forest', 'city', 'county', 'other']:
                province_keywords = self._extract_province_keywords(potential_parent)
                for keyword in province_keywords:
                    if keyword in child_location and len(keyword) >= 2:
                        candidates.append({
                            'location': potential_parent,
                            'priority': parent_priority,
                            'length': len(keyword),
                            'score': 50
                        })
                        break
        
        if candidates:
            best_parent = max(candidates, key=lambda x: (-x['priority'], x['score'], x['length']))
            return best_parent['location']
        
        return None

def _search_nationwide(self, tree_type: str, component: str) -> Optional[Dict]:
        """在全国范围内搜索匹配的方程"""
        if self.verbose:
            print(f"    [全国搜索] 树种: {tree_type}, 组分: {component}")
    
        candidates = []
    
        for location, location_data in self.location_equations.items():
            matching_trees = [
                t for t in location_data.keys()
                if self._is_tree_type_match(tree_type, t)
            ]
        
            if not matching_trees:
                continue
        
            for tree in matching_trees:
                tree_data = location_data[tree]
            
                if component in tree_data:
                    candidates.append({
                        'location': location,
                        'confidence': 0.60,
                        'tree': tree,
                        'component': component
                    })
                    break
            
                for db_comp in tree_data.keys():
                    if self._is_component_similar(component, db_comp):
                        candidates.append({
                            'location': location,
                            'confidence': 0.55,
                            'tree': tree,
                            'component': db_comp
                        })
                        break
    
        if not candidates:
            if self.verbose:
                print(f"    [全国搜索] ❌ 未找到匹配的方程")
            return None
    
        best_candidate = candidates[0]
    
        if self.verbose:
            print(f"    [全国搜索] ✅ 找到 {len(candidates)} 个候选地区")
            print(f"    [全国搜索] 选择: {best_candidate['location']} (置信度: {best_candidate['confidence']:.0%})")
    
        return {
            'location': best_candidate['location'],
            'confidence': best_candidate['confidence']
        }

            province_clean = province.replace('省', '').replace('自治区', '').replace('市', '')
            if len(province_clean) >= 2:
                province_keywords = [province_clean[:2]]
                if len(province_clean) >= 3:
                    province_keywords.append(province_clean[:3])
            if self.verbose:
                print(f"    [查找同省地区] 使用提取的关键字: {province_keywords}")
        
        # 🔥🔥🔥 关键修复：如果关键字为空，直接返回空列表
        if not province_keywords:
            if self.verbose:
                print(f"    [查找同省地区] ⚠️ 无法提取省份关键字，返回空列表")
            return [], None  # ✅ 返回正确类型
        
        # 🔥 灵活方案：动态从方程库提取同省地区（不依赖硬编码）
        same_province_locs = []
        
        # 提取省份核心词（去掉省/市/区等后缀，取前2-3字）
        import re as _re
        _province_core = _re.sub(r"[省市区自治区直辖市]$", "", province).strip()
        _province_core = _province_core[:3] if len(_province_core) >= 3 else _province_core
        
        # 动态匹配所有包含省份核心词的地区
        for db_location in self.location_equations.keys():
            if _province_core and _province_core in db_location:
                same_province_locs.append(db_location)
        print(f'    [调试动态匹配] _province_core={repr(_province_core)}, 匹配到{len(same_province_locs)}个地区: {same_province_locs[:5]}')
        _all_yunnan = [k for k in self.location_equations.keys() if '云南' in k]
        print(f'    [调试所有云南键] {_all_yunnan}')
        
        # 同时保留原有关键词匹配（补充）
        if province_keywords:
            for db_location in self.location_equations.keys():
                if db_location not in same_province_locs:
                    for keyword in province_keywords:
                        if keyword in db_location and len(keyword) >= 2:
                            same_province_locs.append(db_location)
                            break
        
        
        # 从层级结构中补充同省地区
        if province in self.location_hierarchy:
            queue = [province]
            visited = {province}
            
            while queue:
                current = queue.pop(0)
                info = self.location_hierarchy.get(current)
                
                if info:
                    for child in info.get('children', []):
                        if child not in visited:
                            visited.add(child)
                            if child not in same_province_locs:
                                same_province_locs.append(child)
                            queue.append(child)

        # 去重
        same_province_locs = list(set(same_province_locs))
        
        # 🔥🔥🔥 关键修复：如果没有找到同省地区，返回空列表
        if not same_province_locs:
            if self.verbose:
                print(f"    [查找同省地区] ⚠️ 未找到同省地区")
            return [], None  # ✅ 返回正确类型，而不是直接 return
        
        if self.verbose:
            print(f"    [查找同省地区] 找到 {len(same_province_locs)} 个同省地区（排序前）")
            if len(same_province_locs) <= 5:
                print(f"    [查找同省地区] 完整列表: {same_province_locs}")
            else:
                print(f"    [查找同省地区] 前5个: {same_province_locs[:5]}")
        
        # ============================================================
        # 🔥🔥🔥 步骤2：科学排序（按匹配精度打分，不依赖字符串特征）
        # ============================================================
        # 匹配精度评分规则：
        # 100分：完全相同（如查"江苏"，地区也是"江苏"）
        #  90分：查询词是地区的前缀（如查"江苏"，地区是"江苏省"）
        #  80分：单省份简称地区（如"江苏"、"浙江"——省级覆盖，优先于具体县市）
        #  70分：精确省份（含"省"字但不跨省，如"江苏省"）
        #  60分：具体地区（市、县、林场等，如"江苏省盱眙县"）
        #  30分：跨省泛化地区（含"、"或"等地区"，如"湖南、湖北...江苏..."）
        
        def _score_location(loc, query_loc, province):
            # 完全相同：最高优先
            if loc == query_loc:
                return 100
            # 查询词是地区前缀（"江苏" in "江苏省"）
            if query_loc in loc and len(loc) - len(query_loc) <= 2:
                return 90
            # 单省份简称：2-3个汉字，无特殊后缀，无顿号
            # 如"江苏"、"浙江"、"福建"——代表省级覆盖
            is_cross_province = '、' in loc or '等地区' in loc or '等' in loc
            if not is_cross_province:
                has_province_suffix = any(kw in loc for kw in ['省', '自治区', '全省'])
                is_short = len(loc) <= 3
                if is_short and not has_province_suffix:
                    return 80  # 省份简称（如"江苏"、"福建"）
                if has_province_suffix and '市' not in loc and '县' not in loc:
                    return 70  # 精确省份（如"江苏省"、"福建省"）
                return 60  # 具体地区（市、县、林场等）
            return 30  # 跨省泛化地区
        
        # 🔥 核心修复：优先选择"有该树种+组分方程"的地区
        def _has_tree_component(loc, tree, comp):
            """检查该地区是否有指定树种+组分的方程
            location_equations结构: {loc: {tree_type: {component: [eqs]}}}
            """
            loc_data = self.location_equations.get(loc, {})
            if not loc_data:
                return False
            # 遍历该地区所有树种
            for eq_tree, comp_dict in loc_data.items():
                # 树种匹配（包含关系）
                if not tree or not (tree in eq_tree or eq_tree in tree):
                    continue
                # 没有指定组分，只要有该树种就返回True
                if not comp:
                    return True
                # 遍历组分
                for eq_comp in comp_dict.keys():
                    if (comp in eq_comp or eq_comp in comp or
                            comp[:2] in eq_comp or eq_comp[:2] in comp):
                        return True
            return False

        def _count_equations(loc):
            """该地区的方程总数（方程越多说明覆盖越广）"""
            loc_data = self.location_equations.get(loc, {})
            return sum(
                len(eqs)
                for comp_dict in loc_data.values()
                for eqs in comp_dict.values()
            )

        # 获取当前查询的树种和组分（从外层作用域）
        _cur_tree = tree_type if tree_type else ''
        _cur_comp = component if component else ''

        def _score_location_v2(loc):
            # 第一优先：有该树种+组分的方程 → 100分
            if _cur_tree and _cur_comp and _has_tree_component(loc, _cur_tree, _cur_comp):
                return 100 + _count_equations(loc) / 1000
            # 第二优先：有该树种的方程（任意组分）
            if _cur_tree and _has_tree_component(loc, _cur_tree, None):
                return 80 + _count_equations(loc) / 1000
            # 第三优先：方程最多的地区（覆盖最广）
            return _count_equations(loc) / 1000

        scored = [(loc, _score_location_v2(loc)) for loc in same_province_locs]
        print(f"    [调试评分] Top3: {[(loc, f'{s:.2f}') for loc, s in sorted(scored, key=lambda x: x[1], reverse=True)[:3]]}")
        scored.sort(key=lambda x: x[1], reverse=True)
        sorted_locs = [loc for loc, score in scored]
        
        # 兼容后续代码（保留分类变量名但不再用于排序）
        precise_provinces = [loc for loc, s in scored if s >= 70]
        specific_regions  = [loc for loc, s in scored if 30 < s < 70]
        generic_regions   = [loc for loc, s in scored if s <= 30]
        
        if self.verbose:
            print(f"    [地区分类] 精确省份({len(precise_provinces)}): {precise_provinces}")
            print(f"    [地区分类] 具体地区({len(specific_regions)}): {specific_regions[:5]}...")
            print(f"    [地区分类] 泛化地区({len(generic_regions)}): {generic_regions}")
            print(f"    [科学排序] Top5: {sorted_locs[:5]}")
        
        # ============================================================
        # 🔥🔥🔥 步骤3：地理智能排序（新增核心功能）
        # ============================================================
        llm_explanation = None
        
        try:
            # 优先对具体地区进行地理智能排序
            if len(specific_regions) > 1 and self.use_local_model and self.model:
                if self.verbose:
                    print(f"    [地理智能匹配] 使用模型分析地理关系（具体地区）...")
                
                try:
                    # 🔥 关键修复1：内层try-catch
                    ranking_result = self._geographic_intelligent_ranking(
                        query_location=query_location,
                        candidate_locations=specific_regions,
                        province=province
                    )
                    
                    # 🔥 关键修复2：验证返回值类型
                    if ranking_result is None:
                        if self.verbose:
                            print(f"    [地理智能匹配] ⚠️ 返回None，使用兜底排序")
                        raise ValueError("排序方法返回None")
                    
                    ranked_specific, explanation = ranking_result
                    
                    # 验证排序结果
                    if ranked_specific and len(ranked_specific) > 0 and ranked_specific != specific_regions:
                        specific_regions = ranked_specific
                        llm_explanation = explanation
                        if self.verbose:
                            print(f"    [地理智能匹配] ✅ 使用LLM排序结果")
                    else:
                        raise ValueError("排序结果无效")
                
                except Exception as inner_e:
                    # 🔥 关键修复3：内层异常处理
                    if self.verbose:
                        print(f"    [地理智能匹配] ⚠️ LLM排序异常: {inner_e}")
                    
                    # 降级为关键词排序
                    specific_regions = self._fallback_geographic_sorting(
                        query_location, specific_regions, province
                    )
            
            # 如果没有LLM或具体地区只有1个，直接使用关键词排序
            elif len(specific_regions) > 1:
                if self.verbose:
                    print(f"    [关键词排序] LLM未启用，使用关键词排序...")
                
                specific_regions = self._fallback_geographic_sorting(
                    query_location, specific_regions, province
                )
            
            # 🔥 对精确省份也做同样处理
            if len(precise_provinces) > 1 and self.use_local_model and self.model:
                if self.verbose:
                    print(f"    [地理智能匹配] 使用模型分析地理关系（精确省份）...")
                
                try:
                    ranking_result = self._geographic_intelligent_ranking(
                        query_location=query_location,
                        candidate_locations=precise_provinces,
                        province=province
                    )
                    
                    if ranking_result is None:
                        raise ValueError("排序方法返回None")
                    
                    ranked_provinces, explanation = ranking_result
                    
                    if ranked_provinces and len(ranked_provinces) > 0 and ranked_provinces != precise_provinces:
                        precise_provinces = ranked_provinces
                        if not llm_explanation:
                            llm_explanation = explanation
                
                except Exception as inner_e:
                    if self.verbose:
                        print(f"    [地理智能匹配] ⚠️ 省份排序异常: {inner_e}")
                    # 保持原顺序或使用关键词排序
                    pass

        except Exception as e:
            # 🔥 外层异常捕获（最后防线）
            if self.verbose:
                print(f"    [地理智能匹配] ❌ 外层异常: {e}")
                import traceback
                traceback.print_exc()
            
            # 发生异常时，使用关键词排序作为兜底
            if len(specific_regions) > 1:
                try:
                    if self.verbose:
                        print(f"    [关键词排序] 异常后使用关键词兜底...")
                    specific_regions = self._fallback_geographic_sorting(
                        query_location, specific_regions, province
                    )
                except Exception as e2:
                    if self.verbose:
                        print(f"    [关键词排序] ❌ 兜底排序也失败: {e2}")
                    # 保持原始顺序
                    pass
        
        # ============================================================
        # 步骤4：强制优先级排序（省份 > 具体 > 泛化）
        # ============================================================
        sorted_locs = precise_provinces + specific_regions + generic_regions
        
        if self.verbose:
            print(f"    [最终排序] 总数: {len(sorted_locs)}")
            print(f"    [最终排序] Top5: {sorted_locs[:5]}")
        
        # 🔥🔥🔥 关键修复：确保最终返回正确类型
        # ★ Fix2: 防止 LLM 解析失败时 sorted_locs 为 None 导致崩溃
        if sorted_locs is None or not isinstance(sorted_locs, list):
            print('[地理匹配] ⚠️ LLM排序返回None，降级到同省原始列表')
            sorted_locs = same_province_locs if isinstance(same_province_locs, list) else []

        return sorted_locs, llm_explanation

def _parse_geographic_intelligence_response(
        self,
        response: str,
        original_candidates: List[str]
    ) -> Tuple[List[str], Optional[str], Optional[Dict]]:
        """
        解析地理智能推理的响应（增强版 - 修复JSON解析）
        
        Returns:
            (排序后的地区列表, 解释文本, 分析详情)
        """
        try:
            # 🔥🔥🔥 步骤1：强化清理响应
            response = response.strip()
            
            # 移除markdown代码块
            if response.startswith("```json"):
                response = response[7:]
            elif response.startswith("```"):
                response = response[3:]
            if response.endswith("```"):
                response = response[:-3]
            response = response.strip()
            
            # 🔥 移除所有注释（包括行内注释）
            lines = response.split('\n')
            cleaned_lines = []
            for line in lines:
                # 移除 // 注释
                if '//' in line:
                    line = line[:line.index('//')]
                # 移除 # 注释
                if '#' in line:
                    line = line[:line.index('#')]
                
                line = line.rstrip()
                if line:  # 只保留非空行
                    cleaned_lines.append(line)
            
            response = '\n'.join(cleaned_lines)
            
            # 🔥 修复常见JSON错误
            # 1. 移除数组/对象末尾的多余逗号
            response = re.sub(r',(\s*[}\]])', r'\1', response)
            
            # 2. 修复缺少逗号的情况（在 } 和 { 之间，或 ] 和 { 之间）
            response = re.sub(r'}\s*{', '},{', response)
            response = re.sub(r']\s*{', '],{', response)
            
            # 3. 移除控制字符
            response = re.sub(r'[\x00-\x1f\x7f-\x9f]', '', response)
            
            if self.verbose:
                print(f"    [JSON清理] 清理后长度: {len(response)}")
            
            # 🔥🔥🔥 步骤2：提取JSON对象
            start_idx = response.find('{')
            end_idx = response.rfind('}')
            
            if start_idx == -1 or end_idx == -1:
                if self.verbose:
                    print(f"    [地理智能匹配] ⚠️ 未找到JSON对象")
                return original_candidates, None, None
            
            json_str = response[start_idx:end_idx+1]
            
            # 🔥🔥🔥 步骤3：尝试解析JSON（带多次重试）
            result = None
            for attempt in range(3):
                try:
                    result = json.loads(re.sub(r',\s*([}\]])', r'\1', json_str))
                    break  # 解析成功，跳出循环
                except json.JSONDecodeError as e:
                    if self.verbose:
                        print(f"    [JSON解析] 尝试{attempt+1}/3失败: {e}")
                    
                    if attempt == 0:
                        # 第一次失败：尝试修复末尾逗号
                        json_str = re.sub(r',(\s*[}\]])', r'\1', json_str)
                    elif attempt == 1:
                        # 第二次失败：尝试替换单引号为双引号
                        json_str = json_str.replace("'", '"')
                    else:
                        # 第三次失败：打印详细错误
                        if self.verbose:
                            print(f"    [JSON解析] ❌ 所有尝试均失败")
                            print(f"    [JSON解析] 错误位置: 第{e.lineno}行, 第{e.colno}列")
                            print(f"    [JSON解析] 错误附近内容:")
                            # 打印错误附近的内容
                            start = max(0, e.pos - 100)
                            end = min(len(json_str), e.pos + 100)
                            print(f"    {json_str[start:end]}")
                            print(f"    {' ' * (e.pos - start)}^ 错误位置")
                        raise
            
            if not result:
                return original_candidates, None, None
            
            # 🔥🔥🔥 步骤4：提取排序结果
            ranked_locations = result.get('ranked_locations', [])
            explanation = result.get('explanation', '')
            analysis = result.get('analysis', {})
            
            if not ranked_locations or len(ranked_locations) == 0:
                if self.verbose:
                    print(f"    [地理智能匹配] ⚠️ 未返回有效排序")
                return original_candidates, None, None
            
            # 🔥🔥🔥 步骤5：验证并补充排序结果
            valid_ranked = []
            
            for loc in ranked_locations:
                # 精确匹配
                if loc in original_candidates:
                    valid_ranked.append(loc)
                else:
                    # 模糊匹配
                    matched = False
                    for candidate in original_candidates:
                        if loc in candidate or candidate in loc:
                            if candidate not in valid_ranked:
                                valid_ranked.append(candidate)
                                matched = True
                                break
                    
                    if not matched and self.verbose:
                        print(f"    [地理智能匹配] ⚠️ 地区 '{loc}' 不在候选列表中")
            
            # 补充缺失的候选
            for loc in original_candidates:
                if loc not in valid_ranked:
                    valid_ranked.append(loc)
            
            if self.verbose:
                print(f"    [地理智能匹配] ✅ 验证后排序: {valid_ranked[:3]}")
            
            return valid_ranked, explanation, analysis
        
        except json.JSONDecodeError as e:
            if self.verbose:
                print(f"    [地理智能匹配] ⚠️ JSON解析失败（最终）: {e}")
            return original_candidates, None, None
        
        except Exception as e:
            if self.verbose:
                print(f"    [地理智能匹配] ⚠️ 解析异常: {e}")
                import traceback
                traceback.print_exc()
            return original_candidates, None, None

            # 陕西省
            '西安': ['陕西', '秦岭', '关中'],
            '汉中': ['陕西', '秦岭', '巴山'],
        }
        
        # 提取查询地点的核心关键词
        query_core = query_location
        for suffix in ['林场', '保护区', '森林公园', '自然保护区', '国家公园', '市', '县', '区', '地区']:
            query_core = query_core.replace(suffix, '')
        
        if self.verbose:

        # 🔥 计算每个候选地区的得分
        scored_candidates = []

            if query_core in geographic_priority:

                for i, keyword in enumerate(priority_keywords):
                    if keyword in loc:
                        # 优先级越高，得分越高
                        keyword_score = (len(priority_keywords) - i) * 100
                        score += keyword_score
                        reasons.append(f"关键词匹配('{keyword}', +{keyword_score})")
                        break
            
            # 🔥 策略2：山系名称直接匹配（次重要）
            mountain_keywords = ['小兴安岭', '大兴安岭', '长白山', '秦岭', '伏牛山', '吕梁山', '太行山', '桐柏山', '大别山']
            for mountain in mountain_keywords:

                reasons.append(f"包含查询(+50)")
            elif loc in query_core:
                score += 30
                reasons.append(f"被查询包含(+30)")
            
            # 🔥 策略4：地区层级（具体地区 > 省级）
            if any(kw in loc for kw in ['市', '县', '林场', '保护区']):
                score += 10
                reasons.append(f"具体地区(+10)")
            
            # 🔥 策略5：省份名匹配
            if province and (province[:2] in loc or province[:3] in loc):
                score += 5
                reasons.append(f"同省(+5)")
            
            scored_candidates.append({
                'location': loc,
                'score': score,
                'reasons': reasons
            })
        
        # 按得分降序排序
        scored_candidates.sort(key=lambda x: x['score'], reverse=True)
        
        if self.verbose:

                if item['reasons']:
                    print(f"         理由: {'; '.join(item['reasons'])}")
        
        sorted_locs = [item['location'] for item in scored_candidates]
        
        return sorted_locs

def _match_downward(self, location: str, tree_type: str, component: str) -> Optional[Dict]:
        """向下匹配"""
        start_location = self._find_location_in_hierarchy(location)
        if not start_location:
            return None
        
        queue = [(start_location, [start_location])]
        visited = {start_location}
        
        while queue:
            current, path = queue.pop(0)
            
            info = self.location_hierarchy.get(current)
            if not info:
                continue
            
            for child in info.get('children', []):
                if child in visited:
                    continue
                
                visited.add(child)
                child_path = path + [child]
                
                if self._has_matching_equations(child, tree_type, component):

                        'match_type': '向下匹配',

                        'match_path': child_path
                    }
                
                queue.append((child, child_path))

def _haversine(self, lat1, lon1, lat2, lon2):
        """计算两点间球面距离（km）"""
        from math import radians, sin, cos, sqrt, atan2
        R = 6371
        dlat = radians(lat2 - lat1)
        dlon = radians(lon2 - lon1)
        a = sin(dlat/2)**2 + cos(radians(lat1)) * cos(radians(lat2)) * sin(dlon/2)**2
        return R * 2 * atan2(sqrt(a), sqrt(1-a))

        candidates.sort(key=lambda x: x[0])
        top_candidates = candidates[:top_k]

        print(f"  [距离匹配] 同省候选 {len(candidates)} 个，检查最近 {len(top_candidates)} 个:")
        for dist, loc in top_candidates[:5]:
            print(f"    {dist:6.0f}km  {loc}")

        # 步骤4：逐个三重匹配
        for dist, candidate_loc in top_candidates:
            if self._has_matching_equations(candidate_loc, tree_type, component):
                print(f"  [距离匹配] ✅ 三重匹配成功: {candidate_loc} ({dist:.0f}km)")
                return {
                    'query_location': query_location,
                    'matched_location': candidate_loc,
                    'match_type': '距离匹配_同省',
                    'confidence': max(0.7, 1.0 - dist/2000),
                    'distance_km': round(dist),
                    'match_path': [query_location, candidate_loc],
                    'province': user_province,
                    'llm_explanation': f'{query_location}属于{user_province}，最近匹配地区{candidate_loc}(距离约{dist:.0f}km)'
                }

        print(f"  [距离匹配] ⚠️ 同省Top{top_k}内无三重匹配，建议使用通用方程")
        return None
